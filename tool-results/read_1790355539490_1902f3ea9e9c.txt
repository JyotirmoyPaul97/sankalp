     7→// ---------------------------------------------------------------------
     8→// 1. VisualBar — animated horizontal bar for demand/supply/gap comparison
     9→// ---------------------------------------------------------------------
    10→
    11→interface VisualBarProps {
    12→  label: string;
    13→  value: number;
    14→  max?: number;
    15→  tone?: "positive" | "info" | "attention" | "critical" | "neutral";
    16→  showValue?: boolean;
    17→  height?: "sm" | "md" | "lg";
    18→  icon?: React.ReactNode;
    19→}
    20→
    21→const BAR_TONE: Record<string, string> = {
    22→  positive: "bg-status-positive",
    23→  info: "bg-status-info",
    24→  attention: "bg-status-attention",
    25→  critical: "bg-status-critical",
    26→  neutral: "bg-muted-foreground",
    27→};
    28→
    29→const BAR_HEIGHT: Record<string, string> = {
    30→  sm: "h-1.5",
    31→  md: "h-2.5",
    32→  lg: "h-4",
    33→};
    34→
    35→export function VisualBar({ label, value, max = 100, tone = "info", showValue = true, height = "md", icon }: VisualBarProps) {
    36→  const pct = Math.min(100, Math.max(0, (value / max) * 100));
    37→  const [animatedPct, setAnimatedPct] = React.useState(0);
    38→
    39→  React.useEffect(() => {
    40→    const timer = setTimeout(() => setAnimatedPct(pct), 100);
    41→    return () => clearTimeout(timer);
    42→  }, [pct]);
    43→
    44→  return (
    45→    <div className="space-y-1">
    46→      {label ? (
    47→        <div className="flex items-center justify-between text-xs">
    48→          <span className="flex items-center gap-1.5 text-muted-foreground">
    49→            {icon}
    50→            {label}
    51→          </span>
    52→          {showValue ? <span className="tabular-nums font-medium">{Math.round(value)}</span> : null}
    53→        </div>
    54→      ) : null}
    55→      <div className={cn("w-full rounded-full bg-muted overflow-hidden", BAR_HEIGHT[height])}>
    56→        <div
    57→          className={cn("h-full rounded-full transition-all duration-700 ease-out", BAR_TONE[tone])}
    58→          style={{ width: `${animatedPct}%` }}
    59→        />
    60→      </div>
    61→    </div>
    62→  );
    63→}

   156→// 4. ProficiencyLadder — visual proficiency comparison
   157→// ---------------------------------------------------------------------
   158→
   159→const PROFICIENCY_LEVELS = ["AWARENESS", "WORKING", "PROFICIENT", "EXPERT"];
   160→const PROFICIENCY_LABELS: Record<string, string> = {
   161→  AWARENESS: "Awareness", WORKING: "Working", PROFICIENT: "Proficient", EXPERT: "Expert",
   162→  BEGINNER: "Beginner", BASIC: "Basic", INTERMEDIATE: "Intermediate", ADVANCED: "Advanced",
   163→};
   164→
   165→interface ProficiencyLadderProps {
   166→  required?: string;
   167→  current?: string;
   168→  showLabels?: boolean;
   169→}
   170→
   171→export function ProficiencyLadder({ required, current, showLabels = true }: ProficiencyLadderProps) {
   172→  const requiredIdx = required ? PROFICIENCY_LEVELS.indexOf(required) : -1;
   173→  const currentIdx = current ? PROFICIENCY_LEVELS.indexOf(current) : -1;
   174→
   175→  return (
   176→    <div className="space-y-2">
   177→      <div className="flex items-end justify-between gap-1">
   178→        {PROFICIENCY_LEVELS.map((level, i) => {
   179→          const isRequired = i === requiredIdx;
   180→          const isCurrent = i === currentIdx;
   181→          const isBelow = currentIdx >= 0 && i <= currentIdx;
   182→          const isGap = requiredIdx >= 0 && currentIdx >= 0 && i > currentIdx && i <= requiredIdx;
   183→
   184→          return (
   185→            <div key={level} className="flex-1 flex flex-col items-center gap-1">
   186→              {showLabels ? (
   187→                <span className={cn("text-[9px] font-medium", isRequired ? "text-status-critical" : isCurrent ? "text-status-info" : "text-muted-foreground")}>
   188→                  {PROFICIENCY_LABELS[level] ?? level}
   189→                </span>
   190→              ) : null}
   191→              <div
   192→                className={cn(
   193→                  "w-full rounded-t transition-all duration-500",
   194→                  isBelow ? "bg-status-info" : isGap ? "bg-status-attention/50" : "bg-muted",
   195→                  isRequired ? "ring-2 ring-status-critical ring-offset-1" : "",
   196→                )}
   197→                style={{ height: `${20 + i * 12}px` }}
   198→              />
   199→              {isRequired ? <div className="text-[8px] text-status-critical font-bold">REQUIRED</div> : null}
   200→              {isCurrent && !isRequired ? <div className="text-[8px] text-status-info font-bold">YOU</div> : null}
   201→            </div>
   202→          );
   203→        })}
   204→      </div>
   205→    </div>
   206→  );
   207→}

   241→// ---------------------------------------------------------------------
   242→// 6. AnimatedCounter — count-up animation for metrics
   243→// ---------------------------------------------------------------------
   244→
   245→interface AnimatedCounterProps {
   246→  value: number;
   247→  duration?: number;
   248→  className?: string;
   249→}
   250→
   251→export function AnimatedCounter({ value, duration = 1000, className }: AnimatedCounterProps) {
   252→  const [displayValue, setDisplayValue] = React.useState(0);
   253→
   254→  React.useEffect(() => {
   255→    let startTime: number | null = null;
   256→    let frameId: number;
   257→
   258→    const animate = (timestamp: number) => {
   259→      if (!startTime) startTime = timestamp;
   260→      const progress = Math.min((timestamp - startTime) / duration, 1);
   261→      const eased = 1 - Math.pow(1 - progress, 3);
   262→      setDisplayValue(Math.round(value * eased));
   263→      if (progress < 1) {
   264→        frameId = requestAnimationFrame(animate);
   265→      }
   266→    };
   267→
   268→    frameId = requestAnimationFrame(animate);
   269→    return () => cancelAnimationFrame(frameId);
   270→  }, [value, duration]);
   271→
   272→  return <span className={cn("tabular-nums", className)}>{displayValue.toLocaleString()}</span>;
   273→}

   549→// ---------------------------------------------------------------------
   550→// 14. SkillDecisionLoop — the signature KAUSHAL DRISHTI visual
   551→// INDUSTRY DEMAND → ROLE → SKILLS → TRAINING CAPABILITY → SKILL GAP →
   552→// CANDIDATE EVIDENCE → INTERVENTION → OUTCOME ↺
   553→// ---------------------------------------------------------------------
   554→
   555→interface SkillDecisionLoopProps {
   556→  size?: "sm" | "md" | "lg";
   557→  activeStep?: number;
   558→  onStepClick?: (index: number) => void;
   559→  className?: string;
   560→}
   561→
   562→const LOOP_STEPS = [
   563→  { label: "Industry Demand", short: "Demand", tone: "attention", color: "#f59e0b" },
   564→  { label: "Role", short: "Role", tone: "info", color: "#3b82f6" },
   565→  { label: "Skills", short: "Skills", tone: "info", color: "#6366f1" },
   566→  { label: "Training Capability", short: "Training", tone: "info", color: "#0ea5e9" },
   567→  { label: "Skill Gap", short: "Gap", tone: "critical", color: "#ef4444" },
   568→  { label: "Candidate Evidence", short: "Evidence", tone: "positive", color: "#10b981" },
   569→  { label: "Intervention", short: "Action", tone: "info", color: "#8b5cf6" },
   570→  { label: "Outcome", short: "Outcome", tone: "positive", color: "#22c55e" },
   571→];
   572→
   573→export function SkillDecisionLoop({ size = "md", activeStep, onStepClick, className }: SkillDecisionLoopProps) {
   574→  const dim = size === "sm" ? 220 : size === "lg" ? 420 : 320;
   575→  const cx = dim / 2;
   576→  const cy = dim / 2;
   577→  const radius = dim / 2 - 28;
   578→  const stepAngle = (2 * Math.PI) / LOOP_STEPS.length;
   579→  const positions = LOOP_STEPS.map((_, i) => {
   580→    const a = -Math.PI / 2 + i * stepAngle;
   581→    return { x: cx + radius * Math.cos(a), y: cy + radius * Math.sin(a) };
   582→  });
   583→
   584→  return (
   585→    <div className={cn("flex items-center justify-center", className)}>
   586→      <svg viewBox={`0 0 ${dim} ${dim}`} className="w-full h-auto max-w-full" style={{ maxHeight: dim }}>
   587→        <defs>
   588→          <radialGradient id="loop-bg" cx="50%" cy="50%" r="50%">
   589→            <stop offset="0%" stopColor="hsl(var(--primary) / 0.05)" />
   590→            <stop offset="100%" stopColor="transparent" />
   591→          </radialGradient>
   592→        </defs>
   593→        <circle cx={cx} cy={cy} r={radius} fill="url(#loop-bg)" stroke="hsl(var(--border))" strokeWidth="1" strokeDasharray="2 3" opacity="0.5" />
   594→        {/* arrows between steps */}
   595→        {positions.map((p, i) => {
   596→          const next = positions[(i + 1) % positions.length];
   597→          const mx = (p.x + next.x) / 2;
   598→          const my = (p.y + next.y) / 2;
   599→          const dx = next.x - p.x;
   600→          const dy = next.y - p.y;
   601→          const len = Math.sqrt(dx * dx + dy * dy);
   602→          const nx = dx / len;
   603→          const ny = dy / len;
   604→          const startGap = { x: p.x + nx * 16, y: p.y + ny * 16 };
   605→          const endGap = { x: next.x - nx * 16, y: next.y - ny * 16 };
   606→          const isActive = activeStep === i;
   607→          return (
   608→            <g key={i}>
   609→              <line x1={startGap.x} y1={startGap.y} x2={endGap.x} y2={endGap.y} stroke={isActive ? "hsl(var(--primary))" : "hsl(var(--border))"} strokeWidth={isActive ? 1.5 : 0.8} opacity={isActive ? 0.9 : 0.4} />
   610→              {/* arrowhead */}
   611→              <polygon points={`${endGap.x},${endGap.y} ${endGap.x - nx * 4 - ny * 3},${endGap.y - ny * 4 + nx * 3} ${endGap.x - nx * 4 + ny * 3},${endGap.y - ny * 4 - nx * 3}`} fill={isActive ? "hsl(var(--primary))" : "hsl(var(--border))"} opacity={isActive ? 0.9 : 0.5} />
   612→              <circle cx={mx} cy={my} r="0.5" fill="hsl(var(--muted-foreground))" opacity="0.3" />
   613→            </g>
   614→          );
   615→        })}
   616→        {/* loop-back arrow (outcome → demand) */}
   617→        <text x={cx} y={cy - 4} textAnchor="middle" fontSize={size === "sm" ? 7 : size === "lg" ? 11 : 9} fill="hsl(var(--foreground))" fontWeight="700" letterSpacing="0.5">SKILL</text>
   618→        <text x={cx} y={cy + 6} textAnchor="middle" fontSize={size === "sm" ? 7 : size === "lg" ? 11 : 9} fill="hsl(var(--foreground))" fontWeight="700" letterSpacing="0.5">DECISION</text>
   619→        <text x={cx} y={cy + 16} textAnchor="middle" fontSize={size === "sm" ? 7 : size === "lg" ? 11 : 9} fill="hsl(var(--foreground))" fontWeight="700" letterSpacing="0.5">LOOP</text>
   620→        {/* step nodes */}
   621→        {LOOP_STEPS.map((s, i) => {
   622→          const p = positions[i];
   623→          const isActive = activeStep === i;
   624→          return (
   625→            <g key={i} style={{ cursor: onStepClick ? "pointer" : "default" }} onClick={() => onStepClick?.(i)}>
   626→              <circle cx={p.x} cy={p.y} r={isActive ? 16 : 13} fill={isActive ? s.color : "hsl(var(--card))"} stroke={s.color} strokeWidth="2" opacity={isActive ? 1 : 0.9} />
   627→              <text x={p.x} y={p.y + 1} textAnchor="middle" fontSize={size === "sm" ? 6 : 8} fill={isActive ? "#fff" : "hsl(var(--foreground))"} fontWeight="700">{i + 1}</text>
   628→              <text x={p.x} y={p.y + (p.y < cy ? -20 : 26)} textAnchor="middle" fontSize={size === "sm" ? 6 : size === "lg" ? 9 : 7.5} fill="hsl(var(--foreground))" fontWeight={isActive ? 700 : 500}>{s.label}</text>
   629→            </g>
   630→          );
   631→        })}
   632→      </svg>
   633→    </div>
   634→  );
   635→}
