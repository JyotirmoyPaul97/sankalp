   151→function useFilterableGaps(path: string) {
   152→  const [data, setData] = React.useState<{ roles: GapSignal[] } | null>(null);
   153→  const [loading, setLoading] = React.useState(true);
   154→  const [error, setError] = React.useState<{ message: string } | null>(null);
   155→  React.useEffect(() => {
   156→    if (!path) return;
   157→    let cancelled = false;
   158→    setLoading(true);
   159→    fetch(path)
   160→      .then((r) => r.json())
   161→      .then((d) => { if (!cancelled && d.success) { setData(d.data); setLoading(false); } })
   162→      .catch((e) => { if (!cancelled) { setError({ message: e.message }); setLoading(false); } });
   163→    return () => { cancelled = true; };
   164→  }, [path]);
   165→  return { data, loading, error };
   166→}
