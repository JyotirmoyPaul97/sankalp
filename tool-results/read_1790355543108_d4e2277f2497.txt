    18→    18→  icon?: React.ReactNode;

    20→    20→
    21→    21→const BAR_TONE: Record<string, string> = {
    22→    22→  positive: "bg-status-positive",
    23→    23→  info: "bg-status-info",
    24→    24→  attention: "bg-status-attention",

    29→    29→const BAR_HEIGHT: Record<string, string> = {

    34→    34→
    35→    35→export function VisualBar({ label, value, max = 100, tone = "info", showValue = true, height = "md", icon }: VisualBarProps) {
    36→    36→  const pct = Math.min(100, Math.max(0, (value / max) * 100));
    37→    37→  const [animatedPct, setAnimatedPct] = React.useState(0);
    38→    38→
    39→    39→  React.useEffect(() => {
    40→    40→    const timer = setTimeout(() => setAnimatedPct(pct), 100);
    41→    41→    return () => clearTimeout(timer);
    42→    42→  }, [pct]);
    43→    43→
    44→    44→  return (
    45→    45→    <div className="space-y-1">
    46→    46→      {label ? (
    47→    47→        <div className="flex items-center justify-between text-xs">
    48→    48→          <span className="flex items-center gap-1.5 text-muted-foreground">
    49→    49→            {icon}
    50→    50→            {label}
    51→    51→          </span>
    52→    52→          {showValue ? <span className="tabular-nums font-medium">{Math.round(value)}</span> : null}

    54→    54→      ) : null}

    61→    61→    </div>
    62→    62→  );
    63→    63→}

   156→   156→// 4. ProficiencyLadder — visual proficiency comparison
   157→   157→// ---------------------------------------------------------------------
   158→   158→
   159→   159→const PROFICIENCY_LEVELS = ["AWARENESS", "WORKING", "PROFICIENT", "EXPERT"];
   160→   160→const PROFICIENCY_LABELS: Record<string, string> = {
   161→   161→  AWARENESS: "Awareness", WORKING: "Working", PROFICIENT: "Proficient", EXPERT: "Expert",
   162→   162→  BEGINNER: "Beginner", BASIC: "Basic", INTERMEDIATE: "Intermediate", ADVANCED: "Advanced",
   163→   163→};
   164→   164→
   165→   165→interface ProficiencyLadderProps {
   166→   166→  required?: string;
   167→   167→  current?: string;
   168→   168→  showLabels?: boolean;
   169→   169→}
   170→   170→
   171→   171→export function ProficiencyLadder({ required, current, showLabels = true }: ProficiencyLadderProps) {
   172→   172→  const requiredIdx = required ? PROFICIENCY_LEVELS.indexOf(required) : -1;
   173→   173→  const currentIdx = current ? PROFICIENCY_LEVELS.indexOf(current) : -1;
   174→   174→
   175→   175→  return (
   176→   176→    <div className="space-y-2">
   177→   177→      <div className="flex items-end justify-between gap-1">
   178→   178→        {PROFICIENCY_LEVELS.map((level, i) => {
   179→   179→          const isRequired = i === requiredIdx;
   180→   180→          const isCurrent = i === currentIdx;
   181→   181→          const isBelow = currentIdx >= 0 && i <= currentIdx;
   182→   182→          const isGap = requiredIdx >= 0 && currentIdx >= 0 && i > currentIdx && i <= requiredIdx;
   183→   183→
   184→   184→          return (
   185→   185→            <div key={level} className="flex-1 flex flex-col items-center gap-1">
   186→   186→              {showLabels ? (
   187→   187→                <span className={cn("text-[9px] font-medium", isRequired ? "text-status-critical" : isCurrent ? "text-status-info" : "text-muted-foreground")}>
   188→   188→                  {PROFICIENCY_LABELS[level] ?? level}
   189→   189→                </span>
   190→   190→              ) : null}
   191→   191→              <div
   192→   192→                className={cn(
   193→   193→                  "w-full rounded-t transition-all duration-500",
   194→   194→                  isBelow ? "bg-status-info" : isGap ? "bg-status-attention/50" : "bg-muted",
   195→   195→                  isRequired ? "ring-2 ring-status-critical ring-offset-1" : "",
   196→   196→                )}
   197→   197→                style={{ height: `${20 + i * 12}px` }}
   198→   198→              />
   199→   199→              {isRequired ? <div className="text-[8px] text-status-critical font-bold">REQUIRED</div> : null}
   200→   200→              {isCurrent && !isRequired ? <div className="text-[8px] text-status-info font-bold">YOU</div> : null}
   201→   201→            </div>
   202→   202→          );
   203→   203→        })}
   204→   204→      </div>
   205→   205→    </div>
   206→   206→  );
   207→   207→}

   241→   241→// ---------------------------------------------------------------------
   242→   242→// 6. AnimatedCounter — count-up animation for metrics
   243→   243→// ---------------------------------------------------------------------
   244→   244→
   245→   245→interface AnimatedCounterProps {
   246→   246→  value: number;
   247→   247→  duration?: number;
   248→   248→  className?: string;
   249→   249→}
   250→   250→
   251→   251→export function AnimatedCounter({ value, duration = 1000, className }: AnimatedCounterProps) {
   252→   252→  const [displayValue, setDisplayValue] = React.useState(0);
   253→   253→
   254→   254→  React.useEffect(() => {
   255→   255→    let startTime: number | null = null;
   256→   256→    let frameId: number;
   257→   257→
   258→   258→    const animate = (timestamp: number) => {
   259→   259→      if (!startTime) startTime = timestamp;
   260→   260→      const progress = Math.min((timestamp - startTime) / duration, 1);
   261→   261→      const eased = 1 - Math.pow(1 - progress, 3);
   262→   262→      setDisplayValue(Math.round(value * eased));
   263→   263→      if (progress < 1) {
   264→   264→        frameId = requestAnimationFrame(animate);
   265→   265→      }
   266→   266→    };
   267→   267→
   268→   268→    frameId = requestAnimationFrame(animate);
   269→   269→    return () => cancelAnimationFrame(frameId);
   270→   270→  }, [value, duration]);
   271→   271→
   272→   272→  return <span className={cn("tabular-nums", className)}>{displayValue.toLocaleString()}</span>;
   273→   273→}
